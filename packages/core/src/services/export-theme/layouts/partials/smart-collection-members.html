{{- /* The Thread roots a smart collection's conditions select, in its order. The same set and order Jant shows a signed-out reader on the smart collection's page (`smartCollections.toPostFilters` handed to `posts.list`), computed from the exported front matter so the page keeps up with the posts in this repository. Every condition reads the root: match `all` (default, AND) or `any` (OR) across active dimensions format one or a slice of `format` values title `title`; for a quote, `source_name`, which is what Jant stores in the title column year one or a slice of publication years (`date`, in UTC) media the root's own `media` entries: `any`, `none`, or a list of kinds, any one of which matches replies whether the root has published replies visibility one or a slice of `public`, `latest_hidden`, `featured` collection one or a slice of collection slugs Drafts and private roots are exported as Hugo drafts, and Unlisted roots with `build.list: never`, so none reaches here; they are checked again so a `--buildDrafts` preview agrees with Jant. */ -}} {{- $page := .page -}} {{- $feed := .feed -}} {{- $selection := $page.Params.selection | default dict -}} {{- $matchMode := $selection.match | default "all" -}} {{- $roots := where (where site.Pages "Type" "post") "Kind" "section" -}} {{- $members := slice -}} {{- range $roots -}} {{- $post := . -}} {{- $params := .Params -}} {{- $baseOk := and (eq ($params.status | default "published") "published") (in (slice "public" "latest_hidden") ($params.visibility | default "public")) -}} {{- $matches := false -}} {{- if $baseOk -}} {{- $checks := slice -}} {{- if isset $selection "format" -}} {{- $wantedFormats := cond (reflect.IsSlice $selection.format) $selection.format (slice $selection.format) -}} {{- $checks = $checks | append (in $wantedFormats $params.format) -}} {{- end -}} {{- if isset $selection "title" -}} {{- $title := $params.title -}} {{- if eq $params.format "quote" -}}{{- $title = $params.source_name -}}{{- end -}} {{- $checks = $checks | append (eq (gt (len (string ($title | default ""))) 0) $selection.title) -}} {{- end -}} {{- if isset $selection "year" -}} {{- $wantedYears := slice -}} {{- if reflect.IsSlice $selection.year -}} {{- range $selection.year -}}{{- $wantedYears = $wantedYears | append (int .) -}}{{- end -}} {{- else -}} {{- $wantedYears = slice (int $selection.year) -}} {{- end -}} {{- $checks = $checks | append (in $wantedYears $post.Date.UTC.Year) -}} {{- end -}} {{- if isset $selection "media" -}} {{- $media := $params.media | default slice -}} {{- $wanted := $selection.media -}} {{- $mediaMatch := false -}} {{- if reflect.IsSlice $wanted -}} {{- range $media -}}{{- if in $wanted .kind -}}{{- $mediaMatch = true -}}{{- end -}}{{- end -}} {{- else if eq $wanted "any" -}} {{- $mediaMatch = gt (len $media) 0 -}} {{- else -}} {{- $mediaMatch = eq (len $media) 0 -}} {{- end -}} {{- $checks = $checks | append $mediaMatch -}} {{- end -}} {{- if isset $selection "replies" -}} {{- $checks = $checks | append (eq (gt (len $post.Pages) 0) $selection.replies) -}} {{- end -}} {{- if isset $selection "visibility" -}} {{- $wantedVis := cond (reflect.IsSlice $selection.visibility) $selection.visibility (slice $selection.visibility) -}} {{- $visMatch := false -}} {{- range $wantedVis -}} {{- if eq . "featured" -}} {{- if ne ($params.featured_at | default "") "" -}}{{- $visMatch = true -}}{{- end -}} {{- else -}} {{- if eq $params.visibility . -}}{{- $visMatch = true -}}{{- end -}} {{- end -}} {{- end -}} {{- $checks = $checks | append $visMatch -}} {{- end -}} {{- if isset $selection "collection" -}} {{- $wantedCols := cond (reflect.IsSlice $selection.collection) $selection.collection (slice $selection.collection) -}} {{- $inCollection := false -}} {{- range $params.collections -}} {{- if in $wantedCols .slug -}}{{- $inCollection = true -}}{{- end -}} {{- end -}} {{- $checks = $checks | append $inCollection -}} {{- end -}} {{- if eq (len $checks) 0 -}} {{- $matches = true -}} {{- else if eq $matchMode "any" -}} {{- $matches = in $checks true -}} {{- else -}} {{- $matches = not (in $checks false) -}} {{- end -}} {{- end -}} {{- if $matches -}} {{- $activity := $post.Date.Unix -}} {{- with $params.last_activity_at -}}{{- $activity = (time.AsTime .).Unix -}}{{- end -}} {{- $rated := ne $params.rating nil -}} {{- $rating := 0 -}} {{- if $rated -}}{{- $rating = int $params.rating -}}{{- end -}} {{- $members = $members | append (dict "post" $post "id" $params.id "published" $post.Date.Unix "activity" $activity "rated" $rated "rating" $rating ) -}} {{- end -}} {{- end -}} {{- $order := $page.Params.sort_order | default "newest" -}} {{- if and (not $feed) (eq $order "rating_desc") (le (len (where $members "rated" true)) 1) -}} {{- $order = "newest" -}} {{- end -}} {{- /* One string key per root, fixed-width numbers first, so a single sort orders by every key in turn. */ -}} {{- $keyed := slice -}} {{- range $members -}} {{- $key := printf "%020d %s" .activity .id -}} {{- if eq $order "oldest" -}} {{- $key = printf "%020d %s" .published .id -}} {{- else if eq $order "rating_desc" -}} {{- $key = printf "%d %02d %020d %s" (cond .rated 1 0) .rating .activity .id -}} {{- end -}} {{- $keyed = $keyed | append (dict "key" $key "post" .post) -}} {{- end -}} {{- $ordered := slice -}} {{- if $keyed -}} {{- range sort $keyed "key" (cond (eq $order "oldest") "asc" "desc") -}} {{- $ordered = $ordered | append .post -}} {{- end -}} {{- end -}} {{- return $ordered -}}
