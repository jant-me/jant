{{- $paginator := . -}} {{- $total := $paginator.TotalPages -}} {{- $current := $paginator.PageNumber -}} {{- if gt $total 1 -}} {{- /* Mirror main site's getPageNumbers: past seven pages, always seven slots — 1, last, and a three-page window around the current page. The second and second-to-last slots hold a page or 0 (an ellipsis marker), so an ellipsis never stands in for a single page. */ -}} {{- $pages := slice -}} {{- if le $total 7 -}} {{- range seq 1 $total -}} {{- $pages = $pages | append . -}} {{- end -}} {{- else -}} {{- $start := sub $current 1 -}} {{- if lt $start 3 -}}{{- $start = 3 -}}{{- end -}} {{- if gt $start (sub $total 4) -}}{{- $start = sub $total 4 -}}{{- end -}} {{- $end := add $start 2 -}} {{- $pages = $pages | append 1 -}} {{- $pages = $pages | append (cond (eq $start 3) 2 0) -}} {{- range seq $start $end -}} {{- $pages = $pages | append . -}} {{- end -}} {{- $pages = $pages | append (cond (eq $end (sub $total 2)) (sub $total 1) 0) -}} {{- $pages = $pages | append $total -}} {{- end -}}
{{- if $paginator.HasPrev -}} Previous {{- else -}} Previous {{- end -}} {{- range $pages -}} {{- if eq . 0 -}} ... {{- else if eq . $current -}} {{ . }} {{- else -}} {{- $href := "" -}} {{- if eq . 1 -}} {{- $href = ($paginator.First.URL | default "/") -}} {{- else -}} {{- with index $paginator.Pagers (sub . 1) -}} {{- $href = .URL -}} {{- end -}} {{- end -}} {{ . }} {{- end -}} {{- end -}} {{- if $paginator.HasNext -}} Next {{- else -}} Next {{- end -}}
